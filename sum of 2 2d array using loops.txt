#include<stdio.h>
int main() {
    int r,c;
    printf("Enter the number of rows: ");
    scanf("%d",&r);
    printf("Enter the number of columns: ");
    scanf("%d",&c);
    int a[r][c],b[r][c],i,j,sum[r][c];
    printf("Enter the elements of first matrix a:\n");
    for(i=0;i<r;i++) {
        for(j=0;j<c;j++) {
            scanf("%d",&a[i][j]);
        }
    }
    printf("Enter the elements of second matrix b:\n");
    for(i=0;i<r;i++) {
        for(j=0;j<c;j++) {
            scanf("%d",&b[i][j]);
        }
    }
    for(i=0;i<r;i++) {
        for(j=0;j<c;j++) {
            sum[i][j] = a[i][j] + b[i][j];
        }
    }
    printf("Sum of the two matrices:\n");
    for(i=0;i<r;i++) {
        for(j=0;j<c;j++) {
            printf("%d ",sum[i][j]);
        }
        printf("\n");
    }
    return 0;
}
